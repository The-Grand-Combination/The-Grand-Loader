# yes makes a CB a treaty term. Names and descriptions: localisation/00_script_extender.csv.

# Copy of TGC's dismantle_forts: the giver's forts and naval bases are destroyed, in every state that has
# one (all_allowed_states).
treaty_dismantle_forts = {
	sprite_index = 16
	is_triggered_only = yes
	constructing_cb = no
	months = 12

	badboy_factor = 0.5
	prestige_factor = 0
	peace_cost_factor = 5
	penalty_factor = 0

	break_truce_prestige_factor = 1
	break_truce_infamy_factor = 1
	break_truce_militancy_factor = 1

	good_relation_prestige_factor = 1
