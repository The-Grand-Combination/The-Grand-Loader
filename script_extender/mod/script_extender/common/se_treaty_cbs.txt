# Script Extender: treaty CBs (treaty = 1 in script_extender.ini). Appended by the plugin to the
# cb_types.txt the game reads (the mod's, else the game's). Same syntax as cb_types.txt; is_treaty_cb =
# yes makes a CB a treaty term. Names and descriptions: localisation/00_script_extender.csv.

	penalty_factor = 0

		NOT = { war_with = THIS }

		is_colonial = no
